# Source : `uci export network` on the running device (sanitised)
#
# Key points:
#   * wwan  = DHCP client on phy0-sta0, gets 192.168.1.102 from the upstream router
#   * lan   = 192.168.8.0/24 (br-lan bridges eth0.1 + phy0-ap0)
#   * wan/wan6 kept but DISABLED - they point at a dead 172.19.2.254 gateway
#     for an outdoor unit that is not attached. Delete them, or re-enable if you
#     ever reconnect that unit.
#
# The stock `option macaddr` overrides on eth0.1/eth0.2 were removed here: they
# are factory-specific and would be wrong on any other device.

package network

config interface 'loopback'
        option device 'lo'
        option proto 'static'
        option ipaddr '127.0.0.1'
        option netmask '255.0.0.0'

config globals 'globals'
        option ula_prefix 'fd42:3225:f413::/48'

config device
        option name 'br-lan'
        option type 'bridge'
        list ports 'eth0.1'

config interface 'lan'
        option device 'br-lan'
        option proto 'static'
        option ipaddr '192.168.8.1'
        option netmask '255.255.255.0'
        option ip6assign '60'

# --- dead wired uplink (inherited from stock, kept disabled) -----------------
config interface 'wan'
        option device 'eth0.2'
        option proto 'static'
        option ipaddr '172.19.2.2'
        option netmask '255.255.255.0'
        option gateway '172.19.2.254'
        option metric '10'
        option peerdns '0'
        option disabled '1'

config interface 'wan6'
        option device 'eth0.2'
        option proto 'dhcpv6'
        option disabled '1'

# --- switch VLANs (rt305x-esw, cpu @ 6) --------------------------------------
config switch
        option name 'switch0'
        option reset '1'
        option enable_vlan '1'

# VLAN 1 -> LAN ports 0,1,3 + CPU
config switch_vlan
        option device 'switch0'
        option vlan '1'
        option ports '6t 0 1 3'

# VLAN 2 -> WAN ports 2,4 + CPU  (unused while wan is disabled)
config switch_vlan
        option device 'switch0'
        option vlan '2'
        option ports '6t 2 4'

# --- the Wi-Fi uplink ---------------------------------------------------------
config interface 'wwan'
        option device 'phy0-sta0'
        option proto 'dhcp'
